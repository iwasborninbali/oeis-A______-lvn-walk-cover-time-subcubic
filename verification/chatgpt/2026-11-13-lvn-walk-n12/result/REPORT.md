## Method

Starting from K1, append a vertex joined to one, two, or three vertices of degree<3. Every connected graph has a non-cut vertex, so one can delete a vertex and obtain a smaller connected graph; by induction generation is complete. Canonical invariant colors bucket candidates; exact backtracking over bijections preserving all adjacencies eliminates duplicates.

Exact DFS state contains current position and a full array of counters (packed in bytes), and explores all neighbors attaining the minimum counter. The maximal remaining distance to coverage and maximal counter strictly before coverage are both memoized for every state. State counter sum strictly increases on every move, so no recursion cycles occur. Counter overflow (255->256) is checked and would stop the run, but never occurred. A safety guard of 2,000,000 states per graph was never reached in any completed chunk. Complete 0-based witness paths were independently replayed in Python.

```cpp
static void colorize(Graph& g){
```

### walk.cpp

```cpp
// Exact least-visited-neighbour adversarial walk on each generated graph.
// Unbounded by an a priori cap: counter bytes checked for overflow.
// Every legal successor is visited, memoized by (position, entire counters).
// A strictly positive counter-sum increase guarantees acyclic recursion.
#include <bits/stdc++.h>
using namespace std;
using ull=uint64_t;
static ull mix(ull v){v+=0x9e3779b97f4a7c15ULL;v=(v^(v>>30))*0xbf58476d1ce4e5b9ULL;v=(v^(v>>27))*0x94d049bb133111ebULL;return v^(v>>31);}
struct Key {ull lo=0,hi=0;bool operator==(const Key&o)const{return lo==o.lo&&hi==o.hi;}};
struct KeyHash{size_t operator()(const Key&k)const{return mix(k.lo^mix(k.hi));}};
// hi[0..31] = counters of vertices 8..11; [32..39] = position;
// hi[40..51] = positive counters mask (redundant, for speed).
static int ct(Key k,int v){return v<8?(int)((k.lo>>(v*8))&255):(int)((k.hi>>((v-8)*8))&255);}
static int pos(Key k){return (k.hi>>32)&255;}
static int visited(Key k){return (k.hi>>40)&4095;}
static Key step(Key k,int v){
    int old=ct(k,v);
    if(old==255){cerr<<"COUNTER OVERFLOW: cannot continue exact computation\n";exit(3);}
    if(v<8) k.lo+=1ULL<<(v*8);
    else k.hi+=1ULL<<((v-8)*8);
    k.hi=(k.hi&~(255ULL<<32))|((ull)v<<32);
    if(old==0)k.hi|=1ULL<<(40+v);
    return k;
}
static Key initial(int start){Key k;k.hi=((ull)start<<32)|(1ULL<<(40+start));if(start<8)k.lo=1ULL<<(8*start);else k.hi|=1ULL<<((start-8)*8);return k;}
struct Val{uint16_t t=0;uint8_t m=0,nextT=255,nextM=255;};
struct Solver{
    int n=0,all=0;array<uint16_t,12> adj{};
    unordered_map<Key,Val,KeyHash> memo;
    size_t states=0, stateLimit=SIZE_MAX;
    Val solve(Key k){
        if(visited(k)==all)return {};
        auto it=memo.find(k);if(it!=memo.end())return it->second;
        int v=pos(k),minc=999,cm=0;
        for(int i=0;i<n;i++){
            cm=max(cm,ct(k,i));
            if(adj[v]>>i&1)minc=min(minc,ct(k,i));
        }
        Val r;r.m=cm;
        for(int i=0;i<n;i++)if((adj[v]>>i&1)&&ct(k,i)==minc){
            Val sub=solve(step(k,i));
            if(1+sub.t>r.t){r.t=1+sub.t;r.nextT=i;}
            if(sub.m>r.m){r.m=sub.m;r.nextM=i;}
        }
        if(memo.size()>=stateLimit)throw runtime_error("state limit exceeded");
        memo.emplace(k,r);
        return r;
    }
    vector<int> path(int start,bool forM){
        Key k=initial(start);vector<int> p{start};
        while(visited(k)!=all){
            auto it=memo.find(k);
            if(it==memo.end())throw runtime_error("missing memo in path reconstruction");
            int v=forM?it->second.nextM:it->second.nextT;
            if(v==255 && forM)v=it->second.nextT;
            if(v==255)throw runtime_error("dead path");
            k=step(k,v);p.push_back(v);
        }
        return p;
    }
};
static void verify(const array<uint16_t,12>& adj,int n,const vector<int>& p,int expectedT,int expectedM){
```

```python
#!/usr/bin/env python3
"""Independent Python replay of C++ witness paths, including tie legality."""
from pathlib import Path
import re

ROOT = Path(__file__).resolve().parent

def replay(n, index, start, path):
    row = (ROOT / f'graphs_n{n}.txt').read_text().splitlines()[index]
    adjacency = [int(x) for x in row.split()]
    assert len(adjacency) == n and path[0] == start
    counters = [0] * n
    counters[start] = 1
    coverage_step = 0 if n == 1 else None
    largest_before = 0 if n == 1 else 1
    peak_time = 0 if n == 1 else 0
    for t, (u, v) in enumerate(zip(path, path[1:]), 1):
        neighbours = [j for j in range(n) if (adjacency[u] >> j) & 1]
        assert v in neighbours, f'missing edge, step {t}: {u}->{v}'
        assert counters[v] == min(counters[j] for j in neighbours), f'invalid tie, step {t}'
        counters[v] += 1
        assert coverage_step is None, 'path continues after first coverage'
        if all(counters):
            coverage_step = t
        else:
            if max(counters) > largest_before:
                peak_time = t
                largest_before = max(counters)
    assert coverage_step == len(path)-1
    edges = [(i,j) for i in range(n) for j in range(i+1,n) if (adjacency[i] >> j) & 1]
    return coverage_step, largest_before, peak_time, edges, adjacency
```
